    template<typename dTypeIn, typename dTypeOut>
    class DataConvert : public TISP::VirtualOp¶

    Provides node-level APIs for the block copy kernel.

    T

    : Supports float

    dmaDir

    : Supports NO_DMA, DMA_IN, DMA_OUT, and DMA_IN_OUT
